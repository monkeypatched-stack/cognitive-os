# Plan: Fix All 8 TODOs in `bus.py` — Execution Engine Refactor

## Architectural Context

The Cognitive Kernel is a three-subsystem OS:

```
Planner (decides)  →  Execution Engine (executes)  →  Learner (improves)
   Goal/Intent            Workload → Step → Result         EPA/Loss/Policy
   Planning               Provider → Agent → Capability    Knowledge/Utility
   Simulation
   Workload synthesis
```

The execution engine's contract is: `execute(workload) -> ExecutionResult`

It knows **nothing** about NLP, intent classification, goal decomposition, solver mesh, planning policies, or pipeline selection. Internally it performs:

```
Workload → Step → Provider → Broca → Agent → Cerebellum → Capability → CapabilityResult
```

The 8 TODOs in `bus.py` are all symptoms of the bus trying to do planning's job. This refactor makes the execution engine's contract explicit.

```python
class CapabilityBus:
    """Execution engine core — routes workloads through steps to capabilities.

    The execution engine's contract is:
        execute(workload) -> ExecutionResult

    Internally:
        Workload → Step → Provider → Broca → Agent → Cerebellum → Capability

    This class knows nothing about:
        NLP, intent classification, goal decomposition,
        solver mesh, planning policies, pipeline selection.
    """

    def __init__(self, registry: CapabilityRegistry | None = None):
        self._registry = registry or CapabilityRegistry()
        self._agent_bus = AgentBus()
        self._routes: list[ProviderRoute] = []
        self._metrics = BusMetrics()
        if self._registry._cerebellum:
            self._registry._cerebellum.register_transport("local", self._local_handler)
```

### Enhanced `execute()` — agent → provider → capability

Addresses TODOs #5, #6 (provider resolves agent, external agents executed via provider):

```python
async def execute(self, capability_name: str, inputs: dict[str, Any]) -> CapabilityResult:
    cap = self._registry.get(capability_name)
    if not cap:
        return CapabilityResult(success=False, error=f"Unknown: {capability_name}")

    desc = cap.describe()

    # 1. Find agent for this capability
    agent_type = self._agent_bus.resolve_agent(capability_name)

    # 2. If agent has an external provider (A2A execution)
    if agent_type:
        pname = self._agent_bus.get_provider_for(agent_type, capability_name)
        if pname and pname != "local":
            provider = self._get_provider_by_name(pname)
            if provider:
                try:
                    result = await provider.execute(desc, inputs)
                except Exception as e:
                    result = CapabilityResult(success=False, error=str(e))
                self._update_metrics(desc, result)
                return result

    # 3. Modality-based provider or local fallback
    provider = self.resolve_provider(desc.modality)
    try:
        if provider:
            result = await provider.execute(desc, inputs)
        else:
            result = await cap.execute(inputs)
    except Exception as e:
        result = CapabilityResult(success=False, error=f"{capability_name}: {e}")

    self._update_metrics(desc, result)
    return result


def _get_provider_by_name(self, name: str) -> IProvider | None:
    for route in self._routes:
        if route.provider.name == name:
            return route.provider
    return None
```

## Step 4: Overhauled `capability_graph()` — Agent → Cap → Preconditions/Effects

Addresses TODO #7 (graph starts from agent nodes):

```python
def capability_graph(self) -> dict[str, Any]:
    """Agent → capabilities → preconditions/effects.

    The graph mirrors the execution engine's flow:
        Agent provides Capability which requires Preconditions and produces Effects.
    """
    nodes, edges = [], []
    seen: set[str] = set()

    # Agent root nodes
    for agent_type, bindings in self._agent_bus._agent_caps.items():
        aid = f"agent:{agent_type}"
        if aid not in seen:
            nodes.append({"id": aid, "type": "agent", "label": agent_type})
            seen.add(aid)

    # Capability nodes + edges
    for cap_name, icap in self._registry._capabilities.items():
        desc = icap.describe()
        cid = f"capability:{cap_name}"
        if cid not in seen:
            nodes.append(
                {
                    "id": cid,
                    "type": "capability",
                    "label": desc.name,
                    "modality": desc.modality.value,
                    "confidence": desc.confidence,
                }
            )
            seen.add(cid)

        # Agent → Capability
        for at in self._agent_bus._cap_agents.get(cap_name, []):
            edges.append({"from": f"agent:{at}", "to": cid, "type": "provides"})

        # Capability → Preconditions
        for pre in desc.preconditions:
            pid = f"precondition:{pre}"
            if pid not in seen:
                nodes.append({"id": pid, "type": "precondition", "label": pre})
                seen.add(pid)
            edges.append({"from": cid, "to": pid, "type": "requires"})

        # Capability → Effects
        for eff in desc.produces if isinstance(desc.produces, (list, dict)) else []:
            eid = f"effect:{eff}"
            if eid not in seen:
                nodes.append({"id": eid, "type": "effect", "label": eff})
                seen.add(eid)
            edges.append({"from": cid, "to": eid, "type": "produces"})

        # Capability → Enables/Disables
        for en in desc.effects.enable:
            edges.append({"from": cid, "to": f"capability:{en}", "type": "enables"})
        for dis in desc.effects.disable:
            edges.append({"from": cid, "to": f"capability:{dis}", "type": "disables"})

    return {"nodes": nodes, "edges": edges}
```

## Step 5: Add `view_graph()` — Visualization

Addresses TODO #8 (graph visualization):

```python
def view_graph(self, fmt: str = "dict") -> dict[str, Any] | str:
    """Visualize the agent→capability graph.

    fmt="dict"    — structured nodes/edges (API response)
    fmt="text"    — ASCII tree
    fmt="mermaid" — Mermaid diagram syntax
    """
    graph = self.capability_graph()

    if fmt == "dict":
        return graph

    if fmt == "text":
        lines = ["Agents → Capabilities → Preconditions/Effects", "=" * 55]
        for node in graph["nodes"]:
            indent = (
                "  " if node["type"] == "capability" else "    " if node["type"] in ("precondition", "effect") else ""
            )
            lines.append(f"{indent}{node['type']}: {node['label']}")
        for edge in graph["edges"]:
            lines.append(f"  {edge['from']} --[{edge['type']}]--> {edge['to']}")
        return "\n".join(lines)

    if fmt == "mermaid":
        lines = ["graph LR"]
        for edge in graph["edges"]:
            a = edge["from"].replace(":", "_")
            b = edge["to"].replace(":", "_")
            lines.append(f"    {a} -->|{edge['type']}| {b}")
        return "\n".join(lines)

    return graph
```
